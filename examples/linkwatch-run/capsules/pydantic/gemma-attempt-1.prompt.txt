```
from typing import List

import toml
from pydantic import BaseModel, validator


class Site(BaseModel):
    name: str
    url: str

    class Config:
        anystr_strip_whitespace = True

    @validator("name")
    def name_not_empty(cls, value):
        if not value:
            raise ValueError("site name can not be empty")
        return value

    @validator("url")
    def url_has_scheme(cls, value):
        if not value.startswith(("http://", "https://")):
            raise ValueError("url must start with http:// or https://")
        return value


class Settings(BaseModel):
    title: str = "linkwatch"
    output: str = "status.html"
    sites: List[Site] = []

    class Config:
        extra = "forbid"


def load_settings(path="sites.toml"):
    data = toml.load(path)
    return Settings(**data)
```

```
class Handler(BaseHTTPRequestHandler):
    def log_message(self, format, *args):
        pass

    def send_body(self, status, body, content_type, headers=None):
        self.send_response(status)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(len(body)))
        for key, value in (headers or {}).items():
            self.send_header(key, value)
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self):
        if self.path == "/ok":
            self.send_body(200, OK_PAGE.encode("utf-8"), "text/html; charset=utf-8",
                           {"Last-Modified": "Wed, 21 Oct 2015 07:28:00 GMT"})
        elif self.path == "/missing":
            self.send_body(404, b"not found", "text/plain")
        elif self.path == "/redirect":
            self.send_body(302, b"", "text/plain", {"Location": "/ok"})
        elif self.path == "/slow":
            time.sleep(6)
```
