### linkwatch/checker.py
```
import time
from datetime import datetime, timezone

import certifi
import requests
from requests import Session as S

from .dates import parse_last_modified
from .encoding import page_title


def fetch(url):
    session = S()
    try:
        return session.get(url, verify=certifi.where())
    finally:
        session.close()


def check_site(site):
    result = {
        "name": site.name,
        "url": site.url,
        "up": False,
        "status": None,
        "reason": "",
        "title": "",
        "elapsed_ms": None,
        "last_modified": None,
        "checked_at": datetime.now(timezone.utc).isoformat(timespec="seconds"),
    }
    started = time.monotonic()
    try:
        resp = fetch(site.url)
    except requests.exceptions.ConnectionError:
        result["reason"] = "connection error"
        return result
    except requests.exceptions.Timeout:
        result["reason"] = "timed out"
        return result
    result["elapsed_ms"] = int((time.monotonic() - started) * 1000)
    result["status"] = resp.status_code
    result["up"] = resp.ok
    result["reason"] = resp.reason or ""
    result["title"] = page_title(resp)
    modified = parse_last_modified(resp.headers.get("Last-Modified"))
    if modified:
        result["last_modified"] = modified.isoformat()
    return result


def check_all(sites):
    for site in sites:
        yield check_site(site)
```

class Handler(BaseHTTPRequestHandler):
    def log_message(self, format, *args):
        pass

    def send_body(self, status, body, content_type, headers=None):
        self.send_response(status)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(len(body)))
        for key, value in (headers or {}).items():
            self.send_header(key, value)
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self):
        if self.path == "/ok":
            self.send_body(200, OK_PAGE.encode("utf-8"), "text/html; charset=utf-8",
                           {"Last-Modified": "Wed, 21 Oct 2015 07:28:00 GMT"})
        elif self.path == "/missing":
            self.send_body(404, b"not found", "text/plain")
        elif self.path == "/redirect":
            self.send_body(302, b"", "text/plain", {"Location": "/ok"})
        elif self.path == "/slow":
            time.sleep(6)
            self.send_body(200, OK_PAGE.encode("utf-8"), "text/html; charset=utf-8")
        elif self.path == "/latin1":
            self.send_body(200, LATIN1_PAGE.encode("cp1252"), "text/html")
        elif self.path == "/echo-ua":
            ua = self.headers.get("User-Agent", "")
            self.send_body(200, ua.encode("utf-8"), "text/plain; charset=utf-8")
        else:
            self.send_body(404, b"not found", "text/plain")
